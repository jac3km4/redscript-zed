function Flow() {
  const var count: Int32 = 3;
  let i, j: Int32 = 0;
  for (var k: Int32 = 0; k < count; k += 1) {
    if (k == 1) { continue; } else if (k == 2) { break; } else { Ping(); };
  };
  for (i = 0, j = 1; i < 10, j < 20; i += 1, j += 2) { ; }
  for (;;) { break; }
  while (i > 0) { i -= 1; };
  do { i += 1; } while (i < 3);
  switch (i) {
    case 0:
    case 1: { Ping(); break; }
    default: return;
  };
  { Ping(), Pong(); };
}
